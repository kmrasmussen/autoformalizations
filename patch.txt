  have hsplit : ∑ a, (π s) a * advInf M π s a
      = (∑ a ∈ Z, (π s) a * advInf M π s a)
        + ∑ a ∈ Finset.univ \ Z, (π s) a * advInf M π s a := by
    have hu : Z ∪ (Finset.univ \ Z) = (Finset.univ : Finset A) :=
      Finset.union_sdiff_of_subset (Finset.subset_univ Z)
    rw [← hu, Finset.sum_union Finset.disjoint_sdiff]
